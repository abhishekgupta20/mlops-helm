--------------------------------------------------------------------
How to call the API from your local machine
--------------------------------------------------------------------
{{- if contains "NodePort" .Values.service.type }}

  export NODE_PORT=$(kubectl get -n {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "ml-api.fullname" . }})
  export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
  curl http://$NODE_IP:$NODE_PORT

  # If you're on Minikube, this one-liner is easier:
  minikube service {{ include "ml-api.fullname" . }} -n {{ .Release.Namespace }} --url

{{- else if .Values.ingress.enabled }}
{{- range .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}{{ (index .paths 0).path }}
{{- end }}
{{- else }}

  export POD_NAME=$(kubectl get pods -n {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "ml-api.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl port-forward -n {{ .Release.Namespace }} $POD_NAME 8080:{{ .Values.service.targetPort }}
  curl http://127.0.0.1:8080

{{- end }}
